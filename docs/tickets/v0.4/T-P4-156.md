```ts
interface KeyboardShortcutsParams {
  closeTab: (...) => void
  closePane: (...) => void
  splitRight: (...) => void
  splitDown: (...) => void
  addNewTab: (...) => void
}

export function useKeyboardShortcuts(params: KeyboardShortcutsParams): void {
  const chordRef = useRef<{ kind: 'cmd-k'; timer: number } | null>(null)
  useEffect(() => {
    // isModifier, targetIsEditable, clearChord, onKeyDown
    window.addEventListener('keydown', onKeyDown)
    return () => { window.removeEventListener('keydown', onKeyDown); clearChord() }
  }, [params.closeTab, params.closePane, params.splitRight, params.splitDown, params.addNewTab])
}
```
